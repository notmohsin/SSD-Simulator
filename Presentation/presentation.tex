% ── Slide 9: Outcomes / Findings ─────────────────────────────
\contentslide{Outcomes / Findings}{
  \vspace{0.05cm}
  \threecol{
    \metric{10.67\%}{CMT Hit Rate \soft{\small (bounded LRU)}}
  }{
    \metric{4.50\,J}{NAND Energy \soft{\small (vs.\ 35\,mJ DRAM)}}
  }{
    \metric{8,135}{Peak IOPS \soft{\small (128\,MB CMT, blk=128)}}
  }

  \vspace{0.55cm}
  \begin{itemize}
    \item \tagline{Power gap:} NAND program (4.50\,J) is 127$\times$ more energy-intensive than DRAM cache (35\,mJ) --- quantitatively validates RASP-style cache bypass.
    \item \tagline{CMT size matters for writes:} GC cycles fell from 140 $\to$ 7 as CMT grew from 1\,MB to 128\,MB; IOPS rose from 2,039 $\to$ 8,135.
    \item \tagline{Reads are CMT-insensitive:} Hit rate and IOPS were identical across all CMT sizes --- bottleneck is flash read latency, not the mapping layer.
  \end{itemize}
}

% ── Slide 11: Limitations ─────────────────────────────────────
\contentslide{5. Limitations}{
  \twocol{
    \darkbox{
      \tagline{Simulation Fidelity}

      \vspace{0.12cm}
      SimpleSSD models ideal NAND timing. No thermal throttling, electrical latency spikes, or physical cell degradation was observable.
    }

    \vspace{0.28cm}
    \begin{itemize}
      \item GC triggered but not fully stressed
      \item Wear-levelling factor inactive ($-$1)
      \item No real open-channel SSD hardware
    \end{itemize}
  }{
    \darkbox{
      \tagline{Scope Constraint}

      \vspace{0.12cm}
      12 weeks is insufficient to implement and rigorously validate a complete FTL extension. The CMT is a first step.
    }

    \vspace{0.28cm}
    \begin{itemize}
      \item Only LRU eviction policy tested
      \item RASP/CARDR integration not yet done
      \item Environment setup consumed initial weeks
    \end{itemize}
  }
